      function playChime() {
        if (!soundEnabled) return;
        try {
          const AudioCtx = window.AudioContext || window.webkitAudioContext;
          if (!AudioCtx) return;
          const ctx = new AudioCtx();
          const notes = [523.25, 659.25, 783.99];
          notes.forEach((freq, idx) =&gt; {
            const osc = ctx.createOscillator();
            const gain = ctx.createGain();
            osc.type = &#39;sine&#39;;
            osc.frequency.value = freq;
            gain.gain.setValueAtTime(0.001, ctx.currentTime + idx * 0.14);
            gain.gain.exponentialRampToValueAtTime(0.14, ctx.currentTime + idx * 0.14 + 0.03);
            gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + idx * 0.14 + 0.35);
            osc.connect(gain);
            gain.connect(ctx.destination);
            osc.start(ctx.currentTime + idx * 0.14);
            osc.stop(ctx.currentTime + idx * 0.14 + 0.38);
          });
        } catch (e) {}
      }

      function toggleAmbientNoise() {
        try {
          if (ambientOn) {
            if (ambientNode) {
              ambientNode.stop();
              ambientNode.disconnect();
              ambientNode = null;
            }
            if (ambientCtx) {
              ambientCtx.close();
              ambientCtx = null;
            }
            ambientOn = false;
            showToast(&#39;집중 백색소음을 껐어요.&#39;, &#39;info&#39;);
          } else {
            const AudioCtx = window.AudioContext || window.webkitAudioContext;
            if (!AudioCtx) return;
            ambientCtx = new AudioCtx();
            const bufferSize = ambientCtx.sampleRate * 3;
            const buffer = ambientCtx.createBuffer(1, bufferSize, ambientCtx.sampleRate);
            const data = buffer.getChannelData(0);
            let lastOut = 0.0;
            for (let i = 0; i &lt; bufferSize; i++) {
              const white = Math.random() * 2 - 1;
              data[i] = (lastOut + 0.02 * white) / 1.02;
              lastOut = data[i];
              data[i] *= 0.22;
            }
            const source = ambientCtx.createBufferSource();
            source.buffer = buffer;
            source.loop = true;
            const filter = ambientCtx.createBiquadFilter();
            filter.type = &#39;lowpass&#39;;
            filter.frequency.value = 420;
            const gain = ambientCtx.createGain();
            gain.gain.value = 0.25;
            source.connect(filter);
            filter.connect(gain);
            gain.connect(ambientCtx.destination);
            source.start();
            ambientNode = source;
            ambientOn = true;
            showToast(&#39;🎧 부드러운 집중 브라운노이즈 켜짐&#39;, &#39;info&#39;);
          }
          renderAmbientButton();
        } catch (e) {
          ambientOn = false;
          renderAmbientButton();
        }
      }

      function computeTodayRoutinePct() {
        const total = state.routines.length;
        if (total === 0) return 0;
        const done = state.routines.filter((r) =&gt; r.done).length;
        return Math.round((done / total) * 100);

      function computeMonthlyGoalPct() {
        const list = state.monthlyGoals || [];
        if (list.length === 0) return 0;
        const sum = list.reduce((acc, g) =&gt; acc + Math.min(100, Math.max(0, Number(g.progress) || 0)), 0);
        return Math.round(sum / list.length);
      }

      function computeRoadmapPct(cat) {
        const list = (state.roadmaps &amp;&amp; state.roadmaps[cat]) || [];
        if (list.length === 0) return 0;
        const done = list.filter((i) =&gt; i.done).length;
        return Math.round((done / list.length) * 100);
      }

      function computeTierInfo() {
        const routinePct = computeTodayRoutinePct();
        const monthlyPct = computeMonthlyGoalPct();
        const hasRoutines = state.routines.length &gt; 0;
        const hasMonthly = (state.monthlyGoals || []).length &gt; 0;

        let combinedPct = 0;
        if (hasRoutines &amp;&amp; hasMonthly) {
          combinedPct = Math.round((routinePct + monthlyPct) / 2);
        } else if (hasRoutines) {
          combinedPct = routinePct;
        } else if (hasMonthly) {
          combinedPct = monthlyPct;
        }

        let tierIndex = 0;
        for (let i = TIERS.length - 1; i &gt;= 0; i--) {
          if (combinedPct &gt;= TIERS[i].min) {
            tierIndex = i;
            break;
          }
        }
        const current = TIERS[tierIndex];
        const next = tierIndex &lt; TIERS.length - 1 ? TIERS[tierIndex + 1] : null;
        return { routinePct, monthlyPct, combinedPct, tierIndex, current, next };
      }

        const mins = Math.max(1, Math.round(timerTotalSec / 60));

        state.focusMinutesToday += mins;

      // Render Monthly Heatmap Calendar
      function renderMonthlyHeatmap() {
        updateTodayHistory();
        const now = new Date();
        const year = now.getFullYear();
        const month = now.getMonth();
        const todayDate = now.getDate();
        const firstDay = new Date(year, month, 1).getDay();
        const daysInMonth = new Date(year, month + 1, 0).getDate();

        const titleEl = document.getElementById(&#39;heatmap-title&#39;);
        if (titleEl) {
          titleEl.innerHTML = `&lt;i data-lucide=&quot;calendar-days&quot; class=&quot;w-4 h-4 text-emerald-500 shrink-0&quot;&gt;&lt;/i&gt;&lt;span&gt;${year}년 ${month + 1}월 갓생 잔디 캘린더&lt;/span&gt;`;
        }

        const cells = [];
        for (let i = 0; i &lt; firstDay; i++) {
          cells.push(&#39;&lt;div class=&quot;h-9 rounded-lg&quot;&gt;&lt;/div&gt;&#39;);
        }
        for (let day = 1; day &lt;= daysInMonth; day++) {
          const key = `${year}-${String(month + 1).padStart(2, &#39;0&#39;)}-${String(day).padStart(2, &#39;0&#39;)}`;
          const pct = Number(state.historyByDate[key] || 0);
          const isToday = day === todayDate;
          let colorClass = &#39;bg-slate-100 dark:bg-neutral-900 text-slate-400 dark:text-neutral-500 border-slate-200/70 dark:border-neutral-800&#39;;
          if (pct &gt;= 100) {
            colorClass = &#39;bg-emerald-600 dark:bg-emerald-500 text-white border-emerald-600 font-bold&#39;;
          } else if (pct &gt;= 65) {
            colorClass = &#39;bg-emerald-400 dark:bg-emerald-700 text-white border-emerald-400 font-bold&#39;;
          } else if (pct &gt; 0) {
            colorClass = &#39;bg-emerald-200 dark:bg-emerald-950 text-emerald-900 dark:text-emerald-300 border-emerald-300 dark:border-emerald-900 font-semibold&#39;;
          }
          cells.push(`
            &lt;div title=&quot;${key}: ${pct}%&quot; class=&quot;h-9 rounded-lg border flex flex-col items-center justify-center text-xs transition ${colorClass} ${isToday ? &#39;ring-2 ring-brand-500 ring-offset-1 dark:ring-offset-black&#39; : &#39;&#39;}&quot;&gt;
              &lt;span&gt;${day}&lt;/span&gt;
            &lt;/div&gt;
          `);
        }
        document.getElementById(&#39;monthly-heatmap-grid&#39;).innerHTML = cells.join(&#39;&#39;);
        if (window.lucide) lucide.createIcons();
      }

      // Render Weekly Chart
      function renderWeeklyChart() {
        updateTodayHistory();
        const canvas = document.getElementById(&#39;weekly-chart&#39;);
        if (!canvas || typeof Chart === &#39;undefined&#39;) return;

        const labels = [];
        const values = [];
        const dayNames = [&#39;일&#39;, &#39;월&#39;, &#39;화&#39;, &#39;수&#39;, &#39;목&#39;, &#39;금&#39;, &#39;토&#39;];
        const now = new Date();

        for (let i = 6; i &gt;= 0; i--) {
          const d = new Date(now);
          d.setDate(now.getDate() - i);
          const key = formatDateKey(d);
          labels.push(i === 0 ? &#39;오늘&#39; : `${d.getMonth() + 1}/${d.getDate()}(${dayNames[d.getDay()]})`);
          values.push(Number(state.historyByDate[key] || 0));
        }

        const todayPct = values[values.length - 1];
        document.getElementById(&#39;weekly-avg-badge&#39;).textContent = `오늘 ${todayPct}%`;

        const isDark = state.profile.theme === &#39;dark&#39;;
        const gridColor = isDark ? &#39;rgba(163, 163, 163, 0.14)&#39; : &#39;rgba(148, 163, 184, 0.22)&#39;;
        const tickColor = isDark ? &#39;#a3a3a3&#39; : &#39;#64748b&#39;;

        if (weeklyChart) weeklyChart.destroy();

        weeklyChart = new Chart(canvas.getContext(&#39;2d&#39;), {
          type: &#39;bar&#39;,
          data: {
            labels,
            datasets: [
              {
                label: &#39;루틴 달성률 (%)&#39;,
                data: values,
                backgroundColor: values.map((_, idx) =&gt;
                  idx === values.length - 1 ? &#39;#6366f1&#39; : &#39;rgba(99, 102, 241, 0.38)&#39;
                ),
                borderRadius: 8,
                maxBarThickness: 32
              }
            ]
          },
          options: {
            responsive: true,
            maintainAspectRatio: false,
            plugins: {
              legend: { display: false },

              },
              x: { grid: { display: false }, ticks: { color: tickColor } }
            }
          }
        });
      }

            timerRunning = true;
            renderTimerDisplay();
            if (timerInterval) clearInterval(timerInterval);
            timerInterval = setInterval(() =&gt; {
              timerRemainingSec -= 1;
              if (timerRemainingSec &lt;= 0) {
                clearInterval(timerInterval);
                timerInterval = null;
                timerRunning = false;
                timerRemainingSec = 0;
                renderTimerDisplay();
                recordFocusSession();
              } else {
                renderTimerDisplay();
              }
            }, 1000);
          }
